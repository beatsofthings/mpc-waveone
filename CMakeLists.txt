cmake_minimum_required(VERSION 3.16)
project(MPCWaveOne VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(waveone STATIC src/WaveOne.cpp)
target_include_directories(waveone PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(waveone PRIVATE /W4)
else()
  target_compile_options(waveone PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(test_synth tests/test_synth.cpp)
target_link_libraries(test_synth PRIVATE waveone)
add_test(NAME engine_smoke_test COMMAND test_synth)
